% Shared TikZ styles for the hand-drawn diagrams (\input by main.tex).
% Fixed sizes (8pt/7pt) so the diagrams look the same in the 10pt IEEE build
% and the 12pt elsarticle build, where \footnotesize would grow the boxes.
\providecommand{\dfoot}{\fontsize{8}{9.5}\selectfont}
\providecommand{\dscript}{\fontsize{7}{8}\selectfont}
\tikzset{
  >={Stealth[length=2mm]},
  box/.style={draw, rounded corners=2pt, align=center, fill=white,
              inner sep=3pt, font=\dfoot},
  tier/.style={box, fill=blue!6, minimum height=7mm},
  proc/.style={box, fill=gray!8},
  model/.style={box, fill=orange!10},
  dec/.style={draw, diamond, aspect=2.2, align=center, inner sep=1pt,
              fill=yellow!10, font=\dscript},
  res/.style={box, fill=green!8, font=\dscript},
  bad/.style={box, fill=red!8, font=\dscript},
  lab/.style={font=\dscript, inner sep=1pt},
  flow/.style={->, thick},
  aux/.style={->, densely dashed},
}
